struct Vec<T> {
    ptr: *mut T,
    len: usize,
    cap: usize,
}

impl<T> Vec<T> {
    fn new() -> Self {
        Vec {
            ptr: std::ptr::NonNull::dangling().as_ptr(),
            len: 0,
            cap: 0,
        }
    }

    fn push(&mut self, value: T) {
        if self.len == self.cap {
            self.grow();
        }
        unsafe {
            std::ptr::write(self.ptr.add(self.len), value);
        }
        self.len += 1;
    }

    fn pop(&mut self) -> Option<T> {
        if self.len == 0 {
            None
        } else {
            self.len -= 1;
            unsafe { Some(std::ptr::read(self.ptr.add(self.len))) }
        }
    }

    fn grow(&mut self) {
        let new_cap = if self.cap == 0 { 1 } else { self.cap * 2 };
        let new_ptr = if self.cap == 0 {
            unsafe { std::alloc::alloc(std::alloc::Layout::array::<T>(new_cap).unwrap()) }
        } else {
            unsafe {
                std::alloc::realloc(
                    self.ptr as *mut u8,
                    std::alloc::Layout::array::<T>(self.cap).unwrap(),
                    new_cap * std::mem::size_of::<T>(),
                )
            }
        } as *mut T;

        if new_ptr.is_null() {
            std::alloc::handle_alloc_error(std::alloc::Layout::array::<T>(new_cap).unwrap());
        }

        self.ptr = new_ptr;
        self.cap = new_cap;
    }

    fn get(&self, index: usize) -> Option<&T> {
        if index < self.len {
            unsafe { Some(&*self.ptr.add(index)) }
        } else {
            None
        }
    }

    fn len(&self) -> usize {
        self.len
    }

    fn is_empty(&self) -> bool {
        self.len == 0
    }
}

impl<T> Drop for Vec<T> {
    fn drop(&mut self) {
        if self.cap != 0 {
            while let Some(_) = self.pop() {}
            unsafe {
                std::alloc::dealloc(
                    self.ptr as *mut u8,
                    std::alloc::Layout::array::<T>(self.cap).unwrap(),
                );
            }
        }
    }
}

fn main() {
    let mut v = Vec::new();
    v.push(1);
    v.push(2);
    v.push(3);
    
    println!("Length: {}", v.len());
    println!("First: {:?}", v.get(0));
    println!("Last: {:?}", v.pop());
    println!("Length after pop: {}", v.len());
}
