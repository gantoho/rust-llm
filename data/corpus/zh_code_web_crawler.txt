#!/usr/bin/env python3
"""
用 Python 从零实现一个简单的 Web 爬虫。
演示 HTTP 请求、HTML 解析、异步编程和数据存储。
"""

import asyncio
import aiohttp
from bs4 import BeautifulSoup
from urllib.parse import urljoin, urlparse
import json
import csv
from dataclasses import dataclass, asdict
from typing import Set, List, Optional
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

@dataclass
class PageInfo:
    """存储网页信息的数据类"""
    url: str
    title: str
    description: str
    links: List[str]
    status_code: int
    content_length: int

class WebCrawler:
    """异步网页爬虫"""
    
    def __init__(self, max_depth: int = 2, max_concurrent: int = 10):
        self.max_depth = max_depth
        self.max_concurrent = max_concurrent
        self.visited: Set[str] = set()
        self.results: List[PageInfo] = []
        self.semaphore = asyncio.Semaphore(max_concurrent)
    
    def normalize_url(self, url: str) -> str:
        """规范化 URL，去除片段和查询参数"""
        parsed = urlparse(url)
        return f"{parsed.scheme}://{parsed.netloc}{parsed.path}".rstrip('/')
    
    async def fetch_page(self, session: aiohttp.ClientSession, url: str) -> Optional[str]:
        """获取网页内容"""
        async with self.semaphore:
            try:
                async with session.get(url, timeout=aiohttp.ClientTimeout(total=10)) as response:
                    if response.status == 200:
                        return await response.text()
                    return None
            except Exception as e:
                logger.warning(f"Failed to fetch {url}: {e}")
                return None
    
    def parse_html(self, url: str, html: str, status_code: int) -> PageInfo:
        """解析 HTML 提取信息"""
        soup = BeautifulSoup(html, 'html.parser')
        
        title = soup.title.string if soup.title else ''
        description = ''
        meta_desc = soup.find('meta', attrs={'name': 'description'})
        if meta_desc:
            description = meta_desc.get('content', '')
        
        links = []
        for a_tag in soup.find_all('a', href=True):
            absolute_url = urljoin(url, a_tag['href'])
            if absolute_url.startswith('http'):
                links.append(absolute_url)
        
        return PageInfo(
            url=url,
            title=title.strip(),
            description=description.strip(),
            links=links,
            status_code=status_code,
            content_length=len(html)
        )
    
    async def crawl_url(self, session: aiohttp.ClientSession, url: str, depth: int):
        """递归爬取 URL"""
        normalized = self.normalize_url(url)
        if normalized in self.visited or depth > self.max_depth:
            return
        
        self.visited.add(normalized)
        logger.info(f"Crawling (depth={depth}): {normalized}")
        
        html = await self.fetch_page(session, normalized)
        if html is None:
            return
        
        page_info = self.parse_html(normalized, html, 200)
        self.results.append(page_info)
        
        tasks = []
        for link in page_info.links:
            if self.normalize_url(link) not in self.visited:
                tasks.append(self.crawl_url(session, link, depth + 1))
        
        if tasks:
            await asyncio.gather(*tasks)
    
    async def run(self, start_url: str):
        """启动爬虫"""
        async with aiohttp.ClientSession() as session:
            await self.crawl_url(session, start_url, 0)
    
    def save_to_json(self, filename: str):
        """保存结果为 JSON 文件"""
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump([asdict(r) for r in self.results], f, ensure_ascii=False, indent=2)
        logger.info(f"Saved {len(self.results)} pages to {filename}")
    
    def save_to_csv(self, filename: str):
        """保存结果为 CSV 文件"""
        with open(filename, 'w', newline='', encoding='utf-8') as f:
            writer = csv.DictWriter(f, fieldnames=['url', 'title', 'description', 'status_code', 'content_length'])
            writer.writeheader()
            for r in self.results:
                writer.writerow({
                    'url': r.url,
                    'title': r.title,
                    'description': r.description,
                    'status_code': r.status_code,
                    'content_length': r.content_length
                })
        logger.info(f"Saved {len(self.results)} pages to {filename}")

async def main():
    """主函数"""
    crawler = WebCrawler(max_depth=2, max_concurrent=5)
    start_url = "https://example.com"
    
    logger.info(f"Starting crawl from: {start_url}")
    await crawler.run(start_url)
    
    crawler.save_to_json("crawl_results.json")
    crawler.save_to_csv("crawl_results.csv")
    
    logger.info(f"Crawl complete. Visited {len(crawler.visited)} pages, collected {len(crawler.results)} pages.")

if __name__ == '__main__':
    asyncio.run(main())
